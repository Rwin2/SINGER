\begin{align}
M &= \bigl\{\,(x,y)\;\bigm|\;\bm{H}(x,y) > P_{75}(\bm{H})\,\bigr\} \\[8pt]
c_x &= \frac{\sum_{(x,y)\in M} x\,\bm{H}(x,y)}{\sum_{(x,y)\in M} \bm{H}(x,y)} \\[8pt]
c_y &= \frac{\sum_{(x,y)\in M} y\,\bm{H}(x,y)}{\sum_{(x,y)\in M} \bm{H}(x,y)} \\[8pt]
\text{bearing} &= 2\,c_x / W - 1 \\[8pt]
\text{elevation} &= 2\,c_y / H - 1
\end{align}
